addoRel :: (MiniKanren rel var) 
        => Logic Nat var -> Logic Nat var -> Logic Nat var -> Relation rel
addoRel = relation3 "addo" (\x y z -> asum
    [ do
        x <=> zero
        y <=> z
    , fresh2 (\x' z' -> do
        z <=> succ z'
        x <=> succ x'
